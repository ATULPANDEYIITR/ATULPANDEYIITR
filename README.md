# CREATE ONE BOOK / DEGREE
# ============================================================

def create_book(index, degree, x, y):

    color = degree["color"]
    title = degree["title"]
    subtitle = degree["subtitle"]
    university = degree["university"]
    icon = degree["icon"]

    if len(title) > 8:
        title_size = 27
    else:
        title_size = 42

    if max([len(x) for x in subtitle], default=0) > 23:
        subtitle_size = 18
    else:
        subtitle_size = 21

    text = f"""
    <text
        x="{x + 125}"
        y="{y + 55}"
        text-anchor="middle"
        class="title"
        font-size="{title_size}px">
        {html.escape(title)}
    </text>
    """

    subtitle_y = y + 88

    for line in subtitle:

        text += f"""
        <text
            x="{x + 125}"
            y="{subtitle_y}"
            text-anchor="middle"
            class="subtitle"
            font-size="{subtitle_size}px">
            {html.escape(line)}
        </text>
        """

        subtitle_y += 25

    illustration_y = y + 205

    if len(subtitle) >= 4:
        illustration_y = y + 220

    return f"""

    <!-- =================================================
         ACADEMIC BOOK {index}
         ================================================= -->

    <g>

        <!-- slow physical book movement -->

        <animateTransform
            attributeName="transform"
            type="rotate"
            values="
                -0.7 {x + 125} {y + 165};
                 0.7 {x + 125} {y + 165};
                -0.7 {x + 125} {y + 165}"
            dur="{8 + index % 4}s"
            repeatCount="indefinite"/>

        <animateTransform
            attributeName="transform"
            additive="sum"
            type="translate"
            values="
                0 0;
                0 -2;
                0 0;
                0 2;
                0 0"
            dur="{6 + index % 3}s"
            repeatCount="indefinite"/>


        <!-- shadow -->

        <rect
            x="{x + 8}"
            y="{y + 10}"
            width="234"
            height="315"
            rx="16"
            fill="#000000"
            opacity=".65"/>


        <!-- main glass / book -->

        <rect
            x="{x}"
            y="{y}"
            width="250"
            height="315"
            rx="15"
            fill="#07131F"
            stroke="{color}"
            stroke-width="4"
            filter="url(#glow)"/>


        <!-- inner engraved border -->

        <rect
            x="{x + 9}"
            y="{y + 9}"
            width="232"
            height="297"
            rx="11"
            fill="none"
            stroke="{color}"
            stroke-width="1.5"/>


        <!-- corner decoration -->

        <path
            d="
            M{x+18} {y+48}
            V{y+18}
            H{x+48}

            M{x+202} {y+18}
            H{x+232}
            V{y+48}

            H{x+232}
            V{y+267}

            fill="none"
